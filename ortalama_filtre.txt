import cv2
import numpy as np

def mean_filter(image, kernel_size):
    padded_image = cv2.copyMakeBorder(image,
                                      kernel_size // 2,
                                      kernel_size // 2,
                                      kernel_size // 2,
                                      kernel_size // 2,
                                      cv2.BORDER_CONSTANT)

    filtered_image = np.zeros_like(image)

    for y in range(image.shape[0]):
        for x in range(image.shape[1]):
            window = padded_image[y:y + kernel_size, x:x + kernel_size]
            mean_value = np.mean(window)
            filtered_image[y, x] = mean_value

    return filtered_image


image = cv2.imread('5.jpg', cv2.IMREAD_GRAYSCALE)

if image is None:
    print("Hata: Görüntü yüklenemedi. Lütfen dosya adını ve yolunu kontrol edin.")
else:
    kernel_size = 3

    filtered_image = mean_filter(image, kernel_size)

    cv2.imshow('Orijinal Goruntu', image)
    cv2.imshow('Ortalama Filtrelenmis Goruntu', filtered_image)
    cv2.waitKey(0)
    cv2.destroyAllWindows()
